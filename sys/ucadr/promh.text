;;; Set up four pages:
;;;  Page 0: Mapped to the first page of main memory.
;;;  quux: page 0 is mapped to physical page 3, the prom's buffer (contract
;;;  q8).  it is the first of the four pages, 3-6, that the microcode's
;;;  main-memory section (the microcode symbol area) fills, and the prom
;;;  loads that section last, so the buffer is overwritten by what belongs
;;;  there and nothing needs saving.  mit's buffer was physical page 0, which
;;;  on a warm boot is the live world's, so it saved that page to disk block
;;;  1 first and read it back at the end; on a disk with a gpt, block 1 is
;;;  the partition table.
;;;  Page 1: Mapped to the disk control registers.
;;;  Page 2: Mapped to the debug (spy) interface registers at 766000 .
;;;  quux: page 2 is the register page, 17377000 (contract q2), for its word
;;;  102, the mode, in place of unibus 766012.
;;;  Page 3: Mapped to the second page of main memory. (system communication area)
;;; M-A holds the bits to write the level 1 map with r/w access, MD has zero initially.
SET-UP-FOUR-PAGES
	((m-a) dpb m-ones (byte-field 4 22.) a-zero)	;Sets bit 24 which is ignored
	((vma-write-map) add m-a a-3)			;quux: virtual page 0, physical 3

	((M-B) DPB M-ONES (BYTE-FIELD 11 0) A-A)

	((VMA-START-WRITE) DPB M-ONES (BYTE-FIELD 1 9) A-TEMP-1)	;1102
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)

;;; Initialize the disk.
	(CALL DISK-AWAIT-PACK)

;;; In order to not clobber core, save a page (page 0) of physical
;;; memory on disk.  Block 1 is allocated to us for this purpose.
;;; quux: save-a-page, and the 0.1 second delay before its write, are gone:
;;; the buffer is physical page 3, which the microcode's own main-memory
;;; section overwrites, and the prom writes nothing to the disk (contract
;;; q8).

;;; Read the label (block 0) into physical memory page 0, get disk
;;; parameters, and find out where the MICR partition is.
;;; quux: the disk has a gpt, not mit's label (contract q8).  block 0 is
;;; sectors 0 and 1, the protective mbr and the gpt header, so the header
;;; is words 200-377 of the block.  its words 200-201 must be the signature
;;; "EFI PART", 222 (the first sector of the entry array) even and 223
;;; zero, and 225 (an entry's size) 200 bytes, or the prom halts at
;;; error-no-gpt; word 224 is the number of entries.  the entries are read
;;; one block, eight entries, at a time into the buffer, and the first
;;; entry whose type is the microcode's and which has attribute bit 48, the
;;; current bit, is the microcode partition: halt at error-no-current-micr
;;; if there is none, and at error-odd-micr-start if it starts on an odd
;;; sector, which is no block.  the type is told by its first word alone,
;;; which differs between all of q8's types.  mit read the label's word 6,
;;; the current microload's name, and looked for it in the label's
;;; partition table.
;;; quux: into the buffer, physical page 3, virtual page 0.
READ-LABEL
	((M-TEMP-1) SETZ)
	(CALL-XCT-NEXT DISK-READ)
       ((m-a) a-1400)

	;; m-b/ "EFI " = 4022243105
	((m-b) dpb m-ones (byte-field 1 0) a-zero)
	((m-b) dpb m-ones (byte-field 1 2) a-b)
	((m-b) dpb m-ones (byte-field 1 6) a-b)
	((m-b) dpb m-ones (byte-field 2 9.) a-b)
	((m-b) dpb m-ones (byte-field 1 14.) a-b)
	((m-b) dpb m-ones (byte-field 1 16.) a-b)
	((m-b) dpb m-ones (byte-field 1 19.) a-b)
	((m-b) dpb m-ones (byte-field 1 22.) a-b)
	((m-b) dpb m-ones (byte-field 1 29.) a-b)
	((vma-start-read) dpb m-ones (byte-field 1 7) a-zero)	;200
	(jump-if-page-fault error-page-fault)
	(jump-not-equal read-memory-data a-b error-no-gpt)
	;; m-b/ "PART" = 12424440520
	((m-b) dpb m-ones (byte-field 1 4) a-zero)
	((m-b) dpb m-ones (byte-field 1 6) a-b)
	((m-b) dpb m-ones (byte-field 1 8.) a-b)
	((m-b) dpb m-ones (byte-field 1 14.) a-b)
	((m-b) dpb m-ones (byte-field 1 17.) a-b)
	((m-b) dpb m-ones (byte-field 1 20.) a-b)
	((m-b) dpb m-ones (byte-field 1 22.) a-b)
	((m-b) dpb m-ones (byte-field 1 26.) a-b)
	((m-b) dpb m-ones (byte-field 1 28.) a-b)
	((m-b) dpb m-ones (byte-field 1 30.) a-b)
	((vma-start-read) add vma a-1)			;201
	(jump-if-page-fault error-page-fault)
	(jump-not-equal read-memory-data a-b error-no-gpt)
	((m-c) dpb m-ones (byte-field 1 1) a-zero)
	((m-c) dpb m-ones (byte-field 1 4) a-c)
	((vma-start-read) dpb m-ones (byte-field 1 7) a-c)	;222, the entry array's first sector
	(jump-if-page-fault error-page-fault)
	(jump-if-bit-set (byte-field 1 0) read-memory-data error-no-gpt)
	((a-gpt-block) (byte-field 31. 1) read-memory-data)	;its block
	((vma-start-read) add vma a-1)			;223, that sector's high word
	(jump-if-page-fault error-page-fault)
	(jump-not-equal read-memory-data a-zero error-no-gpt)
	((vma-start-read) add vma a-1)			;224, the number of entries
	(jump-if-page-fault error-page-fault)
	((a-gpt-count) read-memory-data)
	((vma-start-read) add vma a-1)			;225, an entry's size
	(jump-if-page-fault error-page-fault)
	((m-temp-1) dpb m-ones (byte-field 1 7) a-zero)	;200 bytes
	(jump-not-equal read-memory-data a-temp-1 error-no-gpt)

	;; m-b/ the microcode's type, 9e318cf5-a95b-4b3b-b2ad-9ae306b0e2da, its
	;; first word 23614306365.
	((m-b) dpb m-ones (byte-field 1 0) a-zero)
	((m-b) dpb m-ones (byte-field 1 2) a-b)
	((m-b) dpb m-ones (byte-field 4 4) a-b)
	((m-b) dpb m-ones (byte-field 2 10.) a-b)
	((m-b) dpb m-ones (byte-field 2 15.) a-b)
	((m-b) dpb m-ones (byte-field 2 20.) a-b)
	((m-b) dpb m-ones (byte-field 4 25.) a-b)
	((m-b) dpb m-ones (byte-field 1 31.) a-b)
	((m-d) setz)					;no entries in the buffer yet

	;; m-b/ the type's first word.
	;; m-c/ the address of an entry in the buffer.
	;; m-d/ the entries left in the buffer.
search-gpt
	(jump-equal m-zero a-gpt-count error-no-current-micr)
	(jump-not-equal m-d a-zero search-gpt-entry)
	((m-temp-1) a-gpt-block)
	(call-xct-next disk-read)
       ((m-a) a-1400)
	((a-gpt-block) m+a+1 m-zero a-gpt-block)
	((m-d) a-10)
	((m-c) setz)
search-gpt-entry
	((vma-start-read) m-c)			;word 0, the type's first word
	(jump-if-page-fault error-page-fault)
	(jump-not-equal read-memory-data a-b search-gpt-next)
	((m-temp-1) dpb m-ones (byte-field 1 3) a-5)	;15, word 13
	((vma-start-read) add m-c a-temp-1)
	(jump-if-page-fault error-page-fault)
	(jump-if-bit-set (byte-field 1 16.) read-memory-data found-partition)
search-gpt-next
	((m-c) add m-c a-40)				;the next entry, 40 words on
	((m-d) sub m-d a-1)
	(jump-xct-next search-gpt)
       ((a-gpt-count) add m-ones a-gpt-count)	;subtract 1

	;; m-c/ the address of the microcode's entry in the buffer.
FOUND-PARTITION
	((vma-start-read) add m-c a-10)		;word 10, the first sector
	(jump-if-page-fault error-page-fault)
	(jump-if-bit-set (byte-field 1 0) read-memory-data error-odd-micr-start)
	((m-a) read-memory-data)
	((A-MICR-ADDRESS) (byte-field 31. 1) read-memory-data)
	((A-MICR-ORIGIN) A-MICR-ADDRESS)
	((m-temp-1) dpb m-ones (byte-field 1 1) a-10)	;12, word 10, the last sector
	((vma-start-read) add m-c a-temp-1)
	(jump-if-page-fault error-page-fault)
	((m-temp-1) sub read-memory-data a-a)		;last - first
	((m-temp-1) m+a+1 m-temp-1 a-zero)		;its sectors
	((A-MICR-N-BLOCKS) (byte-field 31. 1) m-temp-1)	;its blocks
	((M-MICR-OFFSET) A-400)	;So will read in a new page first off

;;; Process one section.  Each section starts with three words:
;;; The section type, the initial address, and the number of locations.
;;; These are gotten into M-B, M-C, and M-D; then the section type is
;;; "dispatched" on.
;;; Section codes are:
;;; 1 = I-MEM, 2 = D-MEM, 3 = MAIN-MEM, 4 = A-M-MEM
PROCESS-SECTION
	(CALL GET-NEXT-WORD)
	(CALL-XCT-NEXT GET-NEXT-WORD)
       ((M-B) M-A)
	(CALL-XCT-NEXT GET-NEXT-WORD)
       ((M-C) M-A)
	((M-D) M-A)
	(JUMP-EQUAL M-B A-1 PROCESS-I-MEM-SECTION)
	(JUMP-EQUAL M-B A-2 PROCESS-D-MEM-SECTION)
	(JUMP-EQUAL M-B A-3 PROCESS-MAIN-MEM-SECTION)
	(JUMP-EQUAL M-B A-4 PROCESS-A-MEM-SECTION)
	(JUMP ERROR-BAD-SECTION-TYPE)

PROCESS-I-MEM-SECTION
	(JUMP-EQUAL M-D A-ZERO PROCESS-SECTION)
	((M-A) (BYTE-FIELD 18. 14.) M-C)
	(JUMP-NOT-EQUAL M-A A-ZERO ERROR-BAD-ADDRESS)
	;; quux: the microcode must end below 36000, where the prom is.
	(JUMP-GREATER-OR-EQUAL M-C A-36000 ERROR-MICROCODE-TOO-BIG)
	(CALL-XCT-NEXT GET-NEXT-WORD)
       ((M-D) SUB M-D A-1)
	(CALL-XCT-NEXT GET-NEXT-WORD)
       ((M-B) M-A)
	;;; Now the first word of the instruction is in A-B, second word is in M-A,
	;;; and the address in I-MEM is in M-C.
	((OA-REG-LOW) DPB M-C (BYTE-FIELD 14. 12.) A-ZERO)
	(WRITE-I-MEM M-A A-B)
	(JUMP-XCT-NEXT PROCESS-I-MEM-SECTION)
       ((M-C) ADD M-C A-1)

PROCESS-D-MEM-SECTION
	(JUMP-EQUAL M-D A-ZERO PROCESS-SECTION)
	((M-A) (BYTE-FIELD 21. 11.) M-C)
	(JUMP-NOT-EQUAL M-A A-ZERO ERROR-BAD-ADDRESS)
	(CALL-XCT-NEXT GET-NEXT-WORD)
       ((M-D) SUB M-D A-1)
	;;; Now M-A has the contents and M-C has the address.
	((OA-REG-LOW) DPB M-C (BYTE-FIELD 11. 12.) A-ZERO)
	(DISPATCH WRITE-DISPATCH-RAM (BYTE-FIELD 0 0) (I-ARG (A-MEM-LOC A-A)))
	(JUMP-XCT-NEXT PROCESS-D-MEM-SECTION)
       ((M-C) ADD M-C A-1)

PROCESS-MAIN-MEM-SECTION
	(CALL GET-NEXT-WORD)
	;;; M-C/ Number of blocks.
	;;; M-D/ Address of first block, relative to beginning of partition.
	;;; M-A/ Physical memory address of first word.
	;; quux: record the section and load it at done-loading, after the
	;; last word has been read through the buffer it overwrites (contract
	;; q8); mit loaded it here.  a section of no blocks, such as the header
	;; that says a base version follows, loads nothing and is passed over.
	;; there is one section with blocks, the microcode symbol area.
	(jump-equal m-c a-zero process-section)
	(jump-not-equal m-zero a-main-mem-n-blocks error-two-main-mem-sections)
	((a-main-mem-n-blocks) m-c)
	((a-main-mem-block) add m-d a-micr-origin)
	(jump-xct-next process-section)
       ((a-main-mem-address) m-a)

PROCESS-A-MEM-SECTION
	((PDL-BUFFER-POINTER) SUB M-C A-1)
A-MEM-LOOP
	(JUMP-EQUAL M-D A-ZERO DONE-LOADING)
	((M-A) (BYTE-FIELD 22. 10.) M-C)
	(JUMP-NOT-EQUAL M-A A-ZERO ERROR-BAD-ADDRESS)
	(CALL-XCT-NEXT GET-NEXT-WORD)
       ((M-D) SUB M-D A-1)
	(JUMP-XCT-NEXT A-MEM-LOOP)
       ((C-PDL-BUFFER-POINTER-PUSH) M-A)

DONE-LOADING
;;; Read back page 0 of physical memory, which we saved on block 1.
;;; quux: nothing was saved (contract q8).  load the main-memory section
;;; recorded by process-main-mem-section, now that nothing more is read
;;; through the buffer, physical page 3.  the section must cover the whole
;;; buffer, or the partition's last block would be left there: halt at
;;; error-buffer-not-loaded unless it starts at or below 1400 and ends at or
;;; above 2000.  the check comes first, so a section that fails it is never
;;; loaded.
	((m-c) a-main-mem-n-blocks)
	(jump-equal m-c a-zero error-buffer-not-loaded)
	((m-a) a-main-mem-address)
	(jump-greater-than m-a a-1400 error-buffer-not-loaded)
	((m-b) dpb m-c (byte-field 22. 8.) a-zero)	;its length in words
	((m-b) add m-b a-main-mem-address)		;the first word past it
	(jump-less-than m-b a-2000 error-buffer-not-loaded)
	((m-b) a-main-mem-block)
main-mem-loop
	(jump-equal m-c a-zero main-mem-loaded)
	(call-xct-next disk-read)
       ((m-temp-1) m-b)
	((m-b) add m-b a-1)
	((m-a) add m-a a-400)
	(jump-xct-next main-mem-loop)
       ((m-c) sub m-c a-1)
main-mem-loaded
	((Q-R) A-400)			;Set up constant needed below.
	;; quux: no md and vma for the prom-disable write; see jump-to-6.

;;; Copy the PDL buffer into A/M memory.
	((PDL-BUFFER-INDEX) SETZ)
FILL-M-LOOP
	((OA-REG-LOW) DPB PDL-BUFFER-INDEX (BYTE-FIELD 5 14.) A-ZERO)
	((M-GARBAGE) C-PDL-BUFFER-INDEX)
	((PDL-BUFFER-INDEX) M+1 PDL-BUFFER-INDEX)
	(JUMP-IF-BIT-CLEAR (BYTE-FIELD 1 5) PDL-BUFFER-INDEX FILL-M-LOOP)
;; quux: copy exactly 2000 words, a memory's size.  mit stopped when the
;; index wrapped to 0, which on a 10-bit cadr is after 1777; on quux's wider
;; index it ran on, and the a address wrapped and zeroed a memory.  stop at
;; 2000, and still at 0 for the cadr, whose index never reaches 2000.
FILL-A-LOOP
	((OA-REG-LOW) DPB PDL-BUFFER-INDEX (BYTE-FIELD 10. 14.) A-ZERO)
	((A-GARBAGE) C-PDL-BUFFER-INDEX)
	((PDL-BUFFER-INDEX) M+1 PDL-BUFFER-INDEX)
	(jump-if-bit-set (byte-field 1 10.) pdl-buffer-index fill-a-done)
	(JUMP-NOT-EQUAL PDL-BUFFER-INDEX A-ZERO FILL-A-LOOP)
fill-a-done
;;; Enter the real world.
;; quux: the prom stays where it is, so it simply jumps to the microcode's
;; location 6, whose loop counts q-r down as mit's prom and ram both did while
;; they waited for the write of 44 to unibus 766012 that disabled the prom.
;;; Note that Q-R is already set up.

JUMP-TO-6
	(JUMP I-MEM-LOC-6)

GET-NEXT-WORD
;;; Get the next word of the MICR partition into M-A.  A-MICR-ADDRESS
;;; contains the number of the next page to be read in from it.
;;; A-MICR-N-BLOCKS has the number of remaining pages in it.
;;; M-MICR-OFFSET has the physical address of the next word.
;;; Clobbers M-TEMP-1, M-TEMP-2 (quux: not Q-R, as nothing divides).
	(JUMP-GREATER-OR-EQUAL M-MICR-OFFSET A-400 GET-NEXT-PAGE)
	((VMA-START-READ) M-MICR-OFFSET)
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)
	(POPJ-AFTER-NEXT (M-A) READ-MEMORY-DATA)
       ((M-MICR-OFFSET) ADD M-MICR-OFFSET A-1)

GET-NEXT-PAGE
	((M-MICR-OFFSET) SETZ)
	(JUMP-GREATER-OR-EQUAL M-ZERO A-MICR-N-BLOCKS ERROR-END-OF-PARTITION)
	((A-MICR-N-BLOCKS) ADD M-ONES A-MICR-N-BLOCKS)	;Subtract 1
	((M-TEMP-1) A-MICR-ADDRESS)
	(CALL-XCT-NEXT DISK-READ)
       ((m-a) a-1400)				;quux: the buffer, physical page 3
	(JUMP-XCT-NEXT GET-NEXT-WORD)
       ((A-MICR-ADDRESS) M+A+1 M-ZERO A-MICR-ADDRESS)

;;; Disk commands.

;;; quux: block-disk has no drive to recalibrate, no seek and no fault to
;;; clear: wait for the controller to be not active, then for a pack
;;; (status <9> is no pack).  mit's disk-recalibrate selected unit 0, cleared
;;; the drive's fault, recalibrated and waited for the drive to be ready.
;;; Clobbers nothing but the VMA and MD.
DISK-AWAIT-PACK
	((VMA-START-READ) A-DISK-REGS)		;Wait for control ready
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)
	(JUMP-IF-BIT-CLEAR (BYTE-FIELD 1 0) READ-MEMORY-DATA DISK-AWAIT-PACK)
	(JUMP-IF-BIT-SET (BYTE-FIELD 1 9.) READ-MEMORY-DATA DISK-AWAIT-PACK)	;No pack
    (ERROR-TABLE AWAIT-DISK-ON-LINE)		;Hangs near here until there is a pack
	(POPJ)

;;; Read one block.
;;; Takes disk block number in M-TEMP-1, phys. mem. address in M-A
;;; Does not clobber M-A.
;;; Clobbers M-TEMP-1, M-TEMP-2 (quux: not Q-R, as nothing divides).
DISK-READ
	(JUMP-XCT-NEXT DISK-OP)
       ((M-TEMP-2) SETZ)

;;; quux: disk-write, which wrote one block, is gone: the prom writes
;;; nothing to the disk (contract q8).  disk-read goes to disk-op itself.

DISK-OP
	(CALL DISK-OP-LOW)
	((M-TEMP-1) AND READ-MEMORY-DATA A-DISK-ERROR)
	(JUMP-NOT-EQUAL M-TEMP-1 A-ZERO ERROR-DISK-ERROR)
	(POPJ)

DISK-OP-LOW
	;; Wait for the disk controller to be ready.
	((VMA-START-READ) A-DISK-REGS)
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)
	(JUMP-IF-BIT-CLEAR (BYTE-FIELD 1 0) READ-MEMORY-DATA DISK-OP)
    (ERROR-TABLE AWAIT-DISK-CONTROL-READY)	;Hangs near here if control hung or absent
	
	;; Write a CCW word to the last loc of sys. comm. area.
	((MD) SELECTIVE-DEPOSIT M-A (BYTE-FIELD 16. 8.) A-ZERO)
	((VMA-START-WRITE) DPB M-ONES (BYTE-FIELD 10. 0) A-ZERO)	;Virt Addr 1777
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)

	;; Write the COMMAND register from M-TEMP-2.
	((MD) M-TEMP-2)
	((VMA-START-WRITE) A-DISK-REGS)
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)
	
	;; Write the CLP register.
	((MD) DPB M-ONES (BYTE-FIELD 9 0) A-ZERO)	;Phys Addr 777 = Virt Addr 1777
	((VMA-START-WRITE) ADD VMA A-1)
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)

DISK-OP-CONTINUED
	;; Write the DISK ADDRESS register.
	;; quux: block-disk's disk address is the block number in M-TEMP-1;
	;; mit's divided it into cylinder, head and sector by the label's
	;; geometry.
	((MD) M-TEMP-1)
	((VMA-START-WRITE) ADD VMA A-1)
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)

	;; Write the START register.
	((VMA-START-WRITE) ADD VMA A-1)
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)

	;; Wait for operation to complete.
DISK-WAIT	
	((VMA-START-READ) A-DISK-REGS)
	(JUMP-IF-PAGE-FAULT ERROR-PAGE-FAULT)
	(JUMP-IF-BIT-CLEAR (BYTE-FIELD 1 0) READ-MEMORY-DATA DISK-WAIT)
    (ERROR-TABLE AWAIT-DISK-DONE)		;Hangs near here while waiting for disk
	(POPJ)

;;; quux: the gpt's error halts (contract q8), here rather than with the
;;; others so that go keeps its address.  each halt is at an even address,
;;; as the others are.
	(no-op)				;so that the halts below are at even addresses
	(no-op)
error-no-gpt
	(jump halt-cons error-no-gpt)
    (error-table error-no-gpt)

	(no-op)
error-no-current-micr
	(jump halt-cons error-no-current-micr)
    (error-table error-no-current-micr)

	(no-op)
error-odd-micr-start
	(jump halt-cons error-odd-micr-start)
    (error-table error-odd-micr-start)
