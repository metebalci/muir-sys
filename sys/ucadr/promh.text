;;; Process one section: its header's type, start and number of items into
;;; m-b, m-c and m-d, then the type "dispatched" on.
process-section
	((m-temp-1) a-sections-left)
	(jump-equal m-temp-1 a-fixnum-zero done-loading)
	((a-sections-left) sub m-temp-1 a-1)
	(call get-section-header)
	(jump-equal m-b a-6 error-section-order)		;section 6 again
	(jump-less-than m-b a-1 error-bad-section-type)
	(jump-greater-than m-b a-4 error-bad-section-type)	;5, or past 6
	(jump-less-or-equal m-b a-last-type error-section-order)	;a type twice, or out of order
	((a-last-type) m-b)
	(jump-equal m-b a-1 process-i-mem-section)
	(jump-equal m-b a-2 process-d-mem-section)
	(jump-equal m-b a-3 process-sym-section)
	(jump process-a-mem-section)

;; the control store: 64 bits an item, <31:0> and then <63:32>, which
;; write-i-mem takes from m<31:0> and a<31:0> (appendix a15b.4).
process-i-mem-section

	(jump-not-equal m-temp-1 a-fix-64 error-bad-width)
	((m-temp-1) a-pad-words)

	(jump-greater-or-equal m-c a-36000 error-microcode-too-big)
	(call-xct-next get-next-word)		;<31:0>
       ((m-d) sub m-d a-1)
	(call-xct-next get-next-word)		;<63:32>
       ((m-b) m-a)
	(call-xct-next skip-padding)
       ((m-e) m-a)
	((oa-reg-low) dpb m-c (byte-field 14. 12.) a-zero)
	(write-i-mem m-b a-e oa-low-select)	;reads oa-reg-low (rev 15)
	(jump-xct-next i-mem-loop)

;; the dispatch memory: 18 bits an item, the entry in <16:0> and its odd

process-d-mem-section
	((m-temp-1) a-section-actual)

	(call-xct-next skip-padding)
       ((m-e) m-a)
	((oa-reg-low) dpb m-c (byte-field 12. 12.) a-zero)
	(dispatch write-dispatch-ram (byte-field 0 0) (i-arg (a-mem-loc a-e)) oa-low-select)
	;the oa-low-select above reads oa-reg-low (rev 15)
	(jump-xct-next d-mem-loop)
       ((m-c) add m-c a-1)

;; the microcode symbol area: 40 bits an item, <31:0> and then <39:32> in
;; <7:0>.  it is page 3, 6000-7777, the buffer the partition is read through,
;; so it is staged in the pdl buffer at 4000-5777, clear of a memory's 0-1777,
;; and copied into place at done-loading, as a memory is.  the buffer stays on
;; page 3: a warm boot keeps main memory, and the symbol area is the one page
;; the prom loads afresh (contract q8); a buffer on any other page would
;; overwrite a page of the world.
process-sym-section
	((m-temp-1) a-section-actual)
	(jump-not-equal m-temp-1 a-fix-40 error-bad-width)
	(jump-not-equal m-c a-fix-6000 error-buffer-not-loaded)
	(jump-not-equal m-d a-fix-2000 error-buffer-not-loaded)
	((m-temp-1) a-pad-words)
	((a-pad-words) sub m-temp-1 a-1)	;two words an item
	((a-sym-loaded) a-1)
	((pdl-buffer-pointer) dpb m-ones (byte-field 11. 0) a-zero)	;3777: pushes from 4000
sym-loop
	(jump-equal m-d a-fixnum-zero process-section)
	(call-xct-next get-next-word)		;<31:0>
       ((m-d) sub m-d a-1)
	(call-xct-next get-next-word)		;<39:32> in <7:0>
       ((m-b) m-a)
	((m-temp-1) (byte-field 24. 8.) m-a)	;<31:8>, padding
	(jump-not-equal m-temp-1 a-zero error-nonzero-padding)
	(call-xct-next skip-padding)
       ((m-e) m-a)
	(jump-xct-next sym-loop)
       ((c-pdl-buffer-pointer-push) dpb m-e (byte-field 8 32.) a-b)

;; a memory: 40 bits an item, as the symbol area's, staged in the pdl buffer
;; from its address and copied at done-loading; the address is checked for
;; every item.
process-a-mem-section
	((m-temp-1) a-section-actual)
	(jump-not-equal m-temp-1 a-fix-40 error-bad-width)
	((m-temp-1) a-pad-words)
	((a-pad-words) sub m-temp-1 a-1)	;two words an item
	((pdl-buffer-pointer) sub m-c a-1)
a-mem-loop
	(jump-equal m-d a-fixnum-zero process-section)
	((m-a) (byte-field 22. 10.) m-c)
	(jump-not-equal m-a a-zero error-bad-address)
	(call-xct-next get-next-word)		;<31:0>
       ((m-d) sub m-d a-1)
	(call-xct-next get-next-word)		;<39:32> in <7:0>
       ((m-b) m-a)
	((m-temp-1) (byte-field 24. 8.) m-a)	;<31:8>, padding
	(jump-not-equal m-temp-1 a-zero error-nonzero-padding)
	(call-xct-next skip-padding)
       ((m-e) m-a)
	((m-c) add m-c a-1)
	(jump-xct-next a-mem-loop)
       ((c-pdl-buffer-pointer-push) dpb m-e (byte-field 8 32.) a-b)

;; a section's 8-word header: the type into m-b, the number of items into m-d,
;; the start into m-c, the widths into a-section-actual and a-section-storage,
;; and the words of padding after an item of one word into a-pad-words.
;; clobbers m-a, m-temp-1 and m-temp-2.
get-section-header
	(call get-next-word)			;0, the type
	(call-xct-next get-next-word)		;1, the number of items
       ((m-b) m-a)
	(call-xct-next get-next-word)		;2, the actual width
       ((m-d) m-a)
	(call-xct-next get-next-word)		;3, the storage width
       ((a-section-actual) m-a)
	(call-xct-next get-next-word)		;4, the start
       ((a-section-storage) m-a)
	(call-xct-next get-next-word)		;5
       ((m-c) m-a)
	(jump-not-equal m-a a-fixnum-zero error-nonzero-padding)
	(call get-next-word)			;6
	(jump-not-equal m-a a-fixnum-zero error-nonzero-padding)
	(call get-next-word)			;7
	(jump-not-equal m-a a-fixnum-zero error-nonzero-padding)
	((m-temp-1) a-section-storage)
	((m-temp-2) (byte-field 5 0) m-temp-1)
	(jump-not-equal m-temp-2 a-zero error-bad-width)	;not a multiple of 32
	(jump-less-than m-temp-1 a-section-actual error-bad-width)	;below the actual
	(popj-after-next (m-temp-2) (byte-field 27. 5) m-temp-1)	;its words
       ((a-pad-words) sub m-temp-2 a-1)

;; the words of padding after an item, a-pad-words of them, each zero.

skip-padding
	((m-hunoz) a-pad-words)
skip-padding-1

	(call get-next-word)
	(jump-not-equal m-a a-fixnum-zero error-nonzero-padding)
	(jump-xct-next skip-padding-1)
       ((m-hunoz) sub m-hunoz a-1)
